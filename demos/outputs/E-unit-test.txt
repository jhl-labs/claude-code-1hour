/* DaVinci NAND controller — DM unit tests on sandbox */

#include <common.h>
#include <dm.h>
#include <dm/test.h>
#include <linux/mtd/rawnand.h>
#include <test/test.h>
#include <test/ut.h>

/* Mock controller register window */
struct mock_emif_regs {
    u32 nandfcr;
    u32 nandfsr;
    u32 nandfecc[4];
    u32 nand4bitecc[4];
    u32 nanderrval1;
    u32 nanderrval2;
};

static struct mock_emif_regs mock;

#define MOCK_NANDFSR_READY  0x1
#define MOCK_4BITECC_BUSY   0xC

/* helpers */
static void mock_reset(void)  { memset(&mock, 0, sizeof(mock)); }
static void mock_ready(void)  { mock.nandfsr = MOCK_NANDFSR_READY; }

/* T1: dev_ready returns 1 when NANDFSR.bit0 set */
static int dm_test_nand_davinci_ready(struct unit_test_state *uts)
{
    struct udevice *dev;
    mock_reset();
    mock_ready();
    ut_assertok(uclass_get_device_by_name(UCLASS_MTD,
                                          "davinci-nand", &dev));
    ut_asserteq(1, nand_davinci_dev_ready(dev_to_mtd(dev)));
    return 0;
}
DM_TEST(dm_test_nand_davinci_ready, UTF_SCAN_FDT);

/* T2: ecc pack — RESERVED bits are masked away */
static int dm_test_nand_davinci_ecc_pack(struct unit_test_state *uts)
{
    /* input  : 0xF???.F???  (bits 28..31 and 12..15 are RESERVED) */
    /* output : 24-bit packed value with reserved bits removed     */
    u32 raw    = 0xFAAAFBBB;            /* hi=0xAAA lo=0xBBB */
    u32 packed = nand_davinci_pack_ecc(raw);
    ut_asserteq(0x00AAABBB, packed);
    return 0;
}
DM_TEST(dm_test_nand_davinci_ecc_pack, 0);

/* T3: 4-bit ECC timeout — controller never leaves BUSY */
static int dm_test_nand_davinci_4bit_timeout(struct unit_test_state *uts)
{
    mock_reset();
    mock.nandfsr = MOCK_4BITECC_BUSY;       /* stuck busy   */
    ut_asserteq(-ETIMEDOUT,
                nand_davinci_4bit_wait_done(&mock));
    return 0;
}
DM_TEST(dm_test_nand_davinci_4bit_timeout, 0);

/* T4: 1-bit error — single flipped bit must be corrected */
static int dm_test_nand_davinci_correct_1bit(struct unit_test_state *uts)
{
    u8 buf[512]; u8 ecc_calc[3], ecc_read[3];
    memset(buf, 0xA5, sizeof(buf));
    nand_davinci_calculate_ecc(NULL, buf, ecc_calc);
    memcpy(ecc_read, ecc_calc, 3);
    buf[42] ^= 0x10;                        /* flip 1 bit   */
    ut_asserteq(1,
        nand_davinci_correct_data(NULL, buf, ecc_read, ecc_calc));
    ut_asserteq(0xA5, buf[42]);             /* fixed back   */
    return 0;
}
DM_TEST(dm_test_nand_davinci_correct_1bit, 0);

/* T5: 3-bit error — uncorrectable, must report -EBADMSG */
static int dm_test_nand_davinci_correct_3bit(struct unit_test_state *uts)
{
    u8 buf[512]; u8 ecc_calc[3], ecc_read[3];
    memset(buf, 0x00, sizeof(buf));
    nand_davinci_calculate_ecc(NULL, buf, ecc_calc);
    memcpy(ecc_read, ecc_calc, 3);
    buf[10] ^= 0x07;                        /* 3 bits flipped */
    ut_asserteq(-EBADMSG,
        nand_davinci_correct_data(NULL, buf, ecc_read, ecc_calc));
    return 0;
}
DM_TEST(dm_test_nand_davinci_correct_3bit, 0);
